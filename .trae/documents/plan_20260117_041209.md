# 每日一字云数据库难度设置优化方案（保证数据一致性）

## 一、方案概述

为确保调整过程中数据与云数据库保持一致，采用以下流程：
1. **完整备份云数据库数据到本地**
2. **在本地进行数据分析和难度调整**
3. **生成调整后的数据文件**
4. **审核调整结果**
5. **更新云数据库**

### 2. 数据分析与难度调整

#### 2.1 编写数据分析脚本

```javascript
// analyze_data.js
const fs = require('fs');
const path = require('path');

const backupFilePath = path.join(__dirname, 'subgames/daily-character/daily_characters_backup.json');
const cloudData = JSON.parse(fs.readFileSync(backupFilePath, 'utf-8'));

// 统计当前难度分布
const counts = {初级: 0, 中级: 0, 高级: 0, 挑战级: 0};
cloudData.forEach(char => {
  counts[char.difficultyLevel] = (counts[char.difficultyLevel] || 0) + 1;
});
console.log('当前难度分布：', counts);

// 统计每个难度级别的占比
const total = cloudData.length;
const percentages = {};
for (let level in counts) {
  percentages[level] = ((counts[level] / total) * 100).toFixed(1) + '%';
}
console.log('当前难度占比：', percentages);

// 查看挑战级字
const challengeChars = cloudData.filter(char => char.difficultyLevel === '挑战级');
console.log('\n挑战级字列表：');
challengeChars.forEach(char => {
  console.log(`  charId: ${char.charId}, char: ${char.char}, pronunciation: ${char.correctPronunciation}`);
});
```

#### 2.3 编写难度调整脚本

```javascript
// adjust_difficulty.js
const fs = require('fs');
const path = require('path');

// 读取备份数据
const backupFilePath = path.join(__dirname, 'subgames/daily-character/daily_characters_backup.json');
const cloudData = JSON.parse(fs.readFileSync(backupFilePath, 'utf-8'));

// 难度调整函数
function adjustDifficulty(character) {
  const char = character.char;
  const charId = character.charId;
  
  // 挑战级字列表（笔画多、生僻）
  const challengeChars = ['纛', '爨', '黟', '龘', '靐', '齉', '龖', '驫', '赑', '屃', '龎', '麤', '鱻', '羴', '犇', '垚', '饕', '餮', '耄', '耋'];
  
  // 高级字列表（较生僻或特殊读音）
  const advancedChars = ['岙', '酂', '疃', '郯', '鄄', '汜', '亳', '郴', '澶', '邗', '崀', '万', '俟', '澹', '桎', '龟', '巷', '乘', '噶', '诃', '溃', '羼', '谮', '臭', '魇'];
  
  // 初级字列表（常见、简单）
  const beginnerChars = ['血', '氛', '种', '处', '当', '桦', '馄', '几', '嫉', '匕', '哄', '脊', '渍', '俭', '贮', '佣', '括', '弦', '岚', '酗'];
  
  // 难度判断
  if (challengeChars.includes(char)) {
    return '挑战级';
  } else if (advancedChars.includes(char)) {
    return '高级';
  } else if (beginnerChars.includes(char)) {
    return '初级';
  } else {
    // 根据字符ID范围分配难度（作为备选方案）
    if (charId <= 138) {
      return '初级';
    } else if (charId <= 359) {
      return '中级';
    } else if (charId <= 497) {
      return '高级';
    } else {
      return '挑战级';
    }
  }
}

// 调整所有字的难度级别
const updatedData = cloudData.map(character => ({
  ...character,
  difficultyLevel: adjustDifficulty(character)
}));

// 统计调整后的分布
const counts = {初级: 0, 中级: 0, 高级: 0, 挑战级: 0};
updatedData.forEach(char => counts[char.difficultyLevel]++);
console.log('调整后难度分布：', counts);

// 生成调整后的数据文件
const outputFilePath = path.join(__dirname, 'subgames/daily-character/daily_characters_optimized.json');
fs.writeFileSync(outputFilePath, JSON.stringify(updatedData, null, 2), 'utf8');
console.log('调整后的数据已保存到:', outputFilePath);

// 生成字、正确读音、难度列表
const listFilePath = path.join(__dirname, 'subgames/daily-character/character_difficulty_list.csv');
const listContent = updatedData.map(char => `${char.char},${char.correctPronunciation},${char.difficultyLevel}`).join('\n');
fs.writeFileSync(listFilePath, '字,正确读音,难度\n' + listContent, 'utf8');
console.log('字、正确读音、难度列表已保存到:', listFilePath);

// 生成仅包含难度调整的更新数据
const updateData = updatedData.map(char => ({
  charId: char.charId,
  difficultyLevel: char.difficultyLevel
}));
const updateFilePath = path.join(__dirname, 'subgames/daily-character/difficulty_update_data.json');
fs.writeFileSync(updateFilePath, JSON.stringify(updateData, null, 2), 'utf8');
console.log('难度更新数据已保存到:', updateFilePath);
```

### 4. 更新云数据库

#### 4.1 使用云函数批量更新

1. **修改云函数**
   ```javascript
   // cloudfunctions/import_to_cloud/index.js
   const cloud = require('wx-server-sdk');
   cloud.init();
   const db = cloud.database();
   const _ = db.command;
   
   exports.main = async (event, context) => {
     const { action } = event;
     
     if (action === 'updateDifficulty') {
       const updateData = require('./difficulty_update_data.json');
       
       // 分批次更新，每批50条
       const batchSize = 50;
       let updatedCount = 0;
       
       for (let i = 0; i < updateData.length; i += batchSize) {
         const batch = updateData.slice(i, i + batchSize);
         
         // 使用事务批量更新
         const transaction = await db.startTransaction();
         
         try {
           for (const item of batch) {
             await transaction.collection('daily_characters')
               .where({ charId: item.charId })
               .update({
                 data: {
                   difficultyLevel: item.difficultyLevel
                 }
               });
           }
           
           await transaction.commit();
           updatedCount += batch.length;
           console.log(`批次 ${i + 1}-${i + batch.length} 更新成功`);
         } catch (error) {
           await transaction.rollback();
           console.error(`批次 ${i + 1}-${i + batch.length} 更新失败:`, error);
           return {
             success: false,
             message: `批次更新失败: ${error.message}`,
             updatedCount: updatedCount
           };
         }
       }
       
       return {
         success: true,
         message: `难度更新完成，共更新 ${updatedCount} 条数据`,
         updatedCount: updatedCount
       };
     }
     
     // 其他功能保持不变...
   };
   ```

## 四、风险控制

1. **数据一致性风险**
   - 解决方案：使用完整备份，确保调整过程中数据与云数据库一致

2. **更新失败风险**
   - 解决方案：分批次更新，每批次使用事务，确保更新的原子性

3. **用户体验风险**
   - 解决方案：在应用中添加难度调整说明，允许用户选择是否使用新的难度设置
